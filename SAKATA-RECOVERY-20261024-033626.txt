async function getPermissionsForMembership(
  membershipId: string,
  organizationId: string
): Promise<Set<string>> {
  const supabase = await createClient();
  const { data: assignments, error: assignmentsError } = await supabase
    .from("membership_roles")
    .select("role_id")
    .eq("membership_id", membershipId);

  if (assignmentsError) {
    throw new Error(
      `Failed to load membership roles: ${assignmentsError.message}`
    );
  }

  if (!assignments || assignments.length === 0) {
    return new Set<string>();
  }

  const roleIds = assignments.map((assignment) => assignment.role_id);

  const { data: roles, error: rolesError } = await supabase
    .from("roles")
    .select("id, organization_id, scope, status")
    .in("id", roleIds)
    .eq("organization_id", organizationId)
    .eq("scope", "organization")
    .eq("status", "active");

  if (rolesError) {
    throw new Error(
      `Failed to load organization roles: ${rolesError.message}`
    );
  }

  if (!roles || roles.length === 0) {
    return new Set<string>();
  }

  const validRoleIds = roles.map((role) => role.id);

  const { data: rolePermissions, error: rolePermissionsError } =
    await supabase
      .from("role_permissions")
      .select("permission_id")
      .in("role_id", validRoleIds);

  if (rolePermissionsError) {
    throw new Error(
      `Failed to load role permissions: ${rolePermissionsError.message}`
    );
  }

  if (!rolePermissions || rolePermissions.length === 0) {
    return new Set<string>();
  }

  const permissionIds = rolePermissions.map(
    (rolePermission) => rolePermission.permission_id
  );

  const { data: permissions, error: permissionsError } = await supabase
    .from("permissions")
    .select("id, code")
    .in("id", permissionIds);

  if (permissionsError) {
    throw new Error(
      `Failed to load permissions: ${permissionsError.message}`
    );
  }

  const permissionSet = new Set<string>();

  for (const permission of permissions ?? []) {
    permissionSet.add(permission.code);
  }

  return permissionSet;
}

export async function getCurrentPermissions(): Promise<Set<string>> {
  const { membership, organization } = await getOrganizationContext();

  if (membership.status !== "active") {
    throw new Error("Forbidden: inactive organization membership.");
  }

  if (organization.status !== "active") {
    throw new Error("Forbidden: inactive organization.");
  }

  return getPermissionsForMembership(membership.id, organization.id);
}

export async function getPermissionsForOrganization(
  organizationId: string
): Promise<Set<string>> {
  const { profile } = await getOrganizationContext();
  const supabase = await createClient();

  const { data: membership, error: membershipError } = await supabase
    .from("organization_memberships")
    .select(
      `
        id,
        organizations!inner (
          status
        )
      `
    )
    .eq("user_profile_id", profile.id)
    .eq("organization_id", organizationId)
    .eq("status", "active")
    .maybeSingle();

  if (membershipError) {
    throw new Error(
      `Failed to resolve organization authorization: ${membershipError.message}`
    );
  }

  const organization = Array.isArray(membership?.organizations)
    ? membership.organizations[0]
    : membership?.organizations;

  if (!membership || organization?.status !== "active") {
    return new Set<string>();
  }

  return getPermissionsForMembership(membership.id, organizationId);
}

export async function hasPermission(
  permissionCode: string
): Promise<boolean> {
  const permissions = await getCurrentPermissions();

  return permissions.has(permissionCode);
}

export async function requirePermission(
  permissionCode: string
): Promise<void> {
  const allowed = await hasPermission(permissionCode);

  if (!allowed) {
    throw new Error(`Forbidden: missing permission ${permissionCode}`);
  }
}

      `Failed to verify active ${productCode} deployment: ${error.message}`
    );
  }

  return Boolean(deployment);
}

export const ACTIVE_ORGANIZATION_COOKIE = "sakata-active-organization";

type Organization = {
  id: string;
  name: string;
  slug: string;
  organization_type: string;
  status: string;
};

type OrganizationMembership = {
  id: string;
  organization_id: string;
  user_profile_id: string;
  status: string;
  joined_at: string;
  organizations: Organization | Organization[] | null;
};

function normalizeOrganization(
  organizations: Organization | Organization[] | null,
): Organization | null {
  if (Array.isArray(organizations)) {
    return organizations[0] ?? null;
  }

  return organizations;
}

async function getAuthenticatedProfile() {
  const supabase = await createClient();

  const {
    data: { user },
    error: userError,
  } = await supabase.auth.getUser();

  if (userError || !user) {
    redirect("/login");
  }

  const { data: profile, error: profileError } = await supabase
    .from("user_profiles")
    .select("id, auth_user_id, display_name, phone, status")
    .eq("auth_user_id", user.id)
    .eq("status", "active")
    .maybeSingle();

  if (profileError) {
    throw new Error(
      `Failed to load authenticated user profile: ${profileError.message}`,
    );
  }

  if (!profile) {
    redirect("/onboarding");
  }

  return {
    supabase,
    user,
    profile,
  };
}

async function getActiveMemberships(
  supabase: Awaited<ReturnType<typeof createClient>>,
  profileId: string,
): Promise<OrganizationMembership[]> {
  const { data: memberships, error: membershipError } = await supabase
    .from("organization_memberships")
    .select(`
      id,
      organization_id,
      user_profile_id,
      status,
      joined_at,
      organizations (
        id,
        name,
        slug,
        organization_type,
        status
      )
    `)
    .eq("user_profile_id", profileId)
    .eq("status", "active")
    .order("joined_at", { ascending: true });

  if (membershipError) {
    throw new Error(
      `Failed to load organization memberships: ${membershipError.message}`,
    );
  }

  return (memberships ?? []) as OrganizationMembership[];
}

export async function getAvailableOrganizations() {
  const { supabase, user, profile } = await getAuthenticatedProfile();

  const memberships = await getActiveMemberships(supabase, profile.id);

  return {
    user,
    profile,
    memberships: memberships
      .map((membership) => ({
        membership,
        organization: normalizeOrganization(membership.organizations),
      }))
      .filter(
        (
          item,
        ): item is {
          membership: OrganizationMembership;
          organization: Organization;
        } =>
          item.organization !== null && item.organization.status === "active",
      ),
  };
}

export async function getOrganizationContext() {
  const { supabase, user, profile } = await getAuthenticatedProfile();

  const memberships = await getActiveMemberships(supabase, profile.id);

  if (memberships.length === 0) {
    redirect("/onboarding");
  }

  /*
   * A single active membership is unambiguous.
   *
   * Do not write a cookie here. Server Components cannot safely mutate
   * response cookies during ordinary rendering.
   */
  if (memberships.length === 1) {
    const membership = memberships[0];
    const organization = normalizeOrganization(membership.organizations);

    if (!organization || organization.status !== "active") {
      redirect("/onboarding");
    }

    return {
      user,
      profile,
      membership,
      organization,
      organizationId: membership.organization_id,
    };
  }

  /*
   * Multiple active organizations require an explicit active-organization
   * selection. The cookie is only a selection hint; the membership above
   * has already been loaded from the authenticated user's database records.
   */
  const cookieStore = await cookies();
  const selectedOrganizationId = cookieStore.get(
    ACTIVE_ORGANIZATION_COOKIE,
  )?.value;

  const membership = selectedOrganizationId
    ? memberships.find(
        (candidate) =>
          candidate.organization_id === selectedOrganizationId,
      )
    : undefined;

  if (!membership) {
    redirect("/organizations/select");
  }

  const organization = normalizeOrganization(membership.organizations);

  if (!organization || organization.status !== "active") {
    redirect("/organizations/select");
  }

  return {
    user,
    profile,
    membership,
    organization,
    organizationId: membership.organization_id,
  };
}

import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const cookieStore = await cookies();

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }) =>
              cookieStore.set(name, value, options)
            );
          } catch {
            // Server Components cannot always write cookies.
          }
        },
      },
    }
  );
}

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function updateSession(request: NextRequest) {
  const sanitizedUrl = request.nextUrl.clone();
  let removedSensitiveQuery = false;

  for (const parameter of Array.from(sanitizedUrl.searchParams.keys())) {
    if (["password", "passwd", "pass"].includes(parameter.toLowerCase())) {
      sanitizedUrl.searchParams.delete(parameter);
      removedSensitiveQuery = true;
    }
  }

  if (sanitizedUrl.pathname === "/login") {
    for (const parameter of Array.from(sanitizedUrl.searchParams.keys())) {
      if (["email", "username"].includes(parameter.toLowerCase())) {
        sanitizedUrl.searchParams.delete(parameter);
        removedSensitiveQuery = true;
      }
    }
  }

  if (removedSensitiveQuery) {
    return NextResponse.redirect(sanitizedUrl);
  }

  let supabaseResponse = NextResponse.next({
    request,
  });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            request.cookies.set(name, value);

            supabaseResponse = NextResponse.next({
              request,
            });

            supabaseResponse.cookies.set(name, value, options);
          });
        },
      },
    }
  );

  const { data } = await supabase.auth.getClaims();

  const claims = data?.claims;

  const isAuthCallback =
    request.nextUrl.pathname === "/auth/callback";

  if (
    !claims &&
    !request.nextUrl.pathname.startsWith("/login") &&
    !isAuthCallback
  ) {
    const url = request.nextUrl.clone();
    url.pathname = "/login";

    return NextResponse.redirect(url);
  }

  return supabaseResponse;
}

import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "@/app/lib/supabase/server";

export async function GET(request: NextRequest) {
  const code = request.nextUrl.searchParams.get("code");

  if (!code) {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  const supabase = await createClient();
  const { error } = await supabase.auth.exchangeCodeForSession(code);

  if (error) {
    return NextResponse.redirect(
      new URL("/login?confirmation=failed", request.url)
    );
  }

  const next = request.nextUrl.searchParams.get("next");
  const destination =
    next === "/auth/recover" ? next : "/auth/continue";

  return NextResponse.redirect(new URL(destination, request.url));
}
